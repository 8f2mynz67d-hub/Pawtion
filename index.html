// Initialize Lucide Icons lucide.createIcons(); const STORAGE_KEY_MULTI = 'pawtion_multi_pets_v2'; const STORAGE_KEY_LEGACY = 'pawtion_pet_profile_v1'; let petStore = { activeId: null, pets: [] }; let currentSpecies = 'cat'; let lastAgeMonths = null; // Life Stage Multipliers (DER Factors) const catStages = [ { id: 'cat_kitten', label: 'ลูกแมว (อายุต่ำกว่า 1 ปี) [DER = 2.5 x RER]', factor: 2.5 }, { id: 'cat_neutered', label: 'แมวโตทำหมันแล้ว (กิจกรรมปกติ) [DER = 1.2 x RER]', factor: 1.2 }, { id: 'cat_intact', label: 'แมวโตยังไม่ทำหมัน [DER = 1.4 x RER]', factor: 1.4 }, { id: 'cat_weight_loss', label: 'แมวโตลดน้ำหนัก / กิจกรรมน้อย [DER = 1.0 x RER]', factor: 1.0 }, { id: 'cat_senior', label: 'แมวสูงวัย (7 ปีขึ้นไป) [DER = 1.1 x RER]', factor: 1.1 }, { id: 'cat_pregnant', label: 'แมวตั้งท้อง / ให้นมลูก [DER = 2.0 x RER]', factor: 2.0 } ]; const dogStages = [ { id: 'dog_puppy_4m', label: 'ลูกสุนัข (อายุต่ำกว่า 4 เดือน) [DER = 3.0 x RER]', factor: 3.0 }, { id: 'dog_puppy_4m_plus', label: 'ลูกสุนัข (อายุ 4 เดือน - 1 ปี) [DER = 2.0 x RER]', factor: 2.0 }, { id: 'dog_neutered', label: 'สุนัขโตทำหมันแล้ว [DER = 1.6 x RER]', factor: 1.6 }, { id: 'dog_intact', label: 'สุนัขโตยังไม่ทำหมัน [DER = 1.8 x RER]', factor: 1.8 }, { id: 'dog_active', label: 'สุนัขแอคทีฟ / ออกกำลังกายเยอะ [DER = 2.0 x RER]', factor: 2.0 }, { id: 'dog_weight_loss', label: 'สุนัขลดน้ำหนัก [DER = 1.0 x RER]', factor: 1.0 }, { id: 'dog_senior', label: 'สุนัขสูงวัย [DER = 1.2 x RER]', factor: 1.2 } ]; // Create a default pet profile object function createDefaultPet(customName = 'น้องม่อน') { return { id: 'pet_' + Date.now() + '_' + Math.floor(Math.random() * 1000), species: 'cat', healthConfirmed: false, petName: customName, petWeight: '4.5', petAgeYears: '2', petAgeMonths: '0', stageId: 'cat_neutered', mainFoodRatio: '70', dryCal: '3.8', wetCal: '0.85', mealsPerDay: '3' }; } // Save all pet profiles to LocalStorage function saveStore() { try { localStorage.setItem(STORAGE_KEY_MULTI, JSON.stringify(petStore)); } catch (e) { console.warn('ไม่สามารถบันทึกข้อมูลลง LocalStorage ได้:', e); } } // Load store from LocalStorage (with automatic migration from v1) function initPetStore() { try { const rawMulti = localStorage.getItem(STORAGE_KEY_MULTI); if (rawMulti) { const parsed = JSON.parse(rawMulti); if (parsed && Array.isArray(parsed.pets) && parsed.pets.length > 0) { petStore = parsed; if (!petStore.pets.some(p => p.id === petStore.activeId)) { petStore.activeId = petStore.pets[0].id; } return; } } // Check legacy single-pet data and migrate const rawLegacy = localStorage.getItem(STORAGE_KEY_LEGACY); if (rawLegacy) { const legacy = JSON.parse(rawLegacy); const migratedPet = { ...createDefaultPet(legacy.petName || 'น้องม่อน'), ...legacy }; petStore = { activeId: migratedPet.id, pets: [migratedPet] }; saveStore(); return; } } catch (e) { console.warn('ไม่สามารถโหลดข้อมูลจาก LocalStorage ได้:', e); } // Fallback: Create initial pet profile const initialPet = createDefaultPet('น้องม่อน'); petStore = { activeId: initialPet.id, pets: [initialPet] }; saveStore(); } // Get currently active pet object function getActivePet() { return petStore.pets.find(p => p.id === petStore.activeId) || petStore.pets[0]; } // Sync current form inputs into the active pet object and save function syncCurrentFormToActivePet() { const activePet = getActivePet(); if (!activePet) return; const stageSelect = document.getElementById('petStage'); const selectedOption = stageSelect.options[stageSelect.selectedIndex]; activePet.species = currentSpecies; activePet.healthConfirmed = document.getElementById('healthConfirmed').checked; activePet.petName = document.getElementById('petName').value; activePet.petWeight = document.getElementById('petWeight').value; activePet.petAgeYears = document.getElementById('petAgeYears').value; activePet.petAgeMonths = document.getElementById('petAgeMonths').value; activePet.stageId = selectedOption ? selectedOption.dataset.id : ''; activePet.mainFoodRatio = document.getElementById('mainFoodRatio').value; activePet.dryCal = document.getElementById('dryCal').value; activePet.wetCal = document.getElementById('wetCal').value; activePet.mealsPerDay = document.getElementById('mealsPerDay').value; saveStore(); renderProfileChips(); } // Load a specific pet profile into the UI form function loadPetIntoForm(petId) { petStore.activeId = petId; saveStore(); const pet = getActivePet(); if (!pet) return; currentSpecies = pet.species === 'dog' ? 'dog' : 'cat'; document.getElementById('healthConfirmed').checked = !!pet.healthConfirmed; document.getElementById('petName').value = pet.petName ?? ''; document.getElementById('petWeight').value = pet.petWeight ?? '4.5'; document.getElementById('petAgeYears').value = pet.petAgeYears ?? '1'; document.getElementById('petAgeMonths').value = pet.petAgeMonths ?? '0'; document.getElementById('mainFoodRatio').value = pet.mainFoodRatio ?? '70'; document.getElementById('dryCal').value = pet.dryCal ?? '3.8'; document.getElementById('wetCal').value = pet.wetCal ?? '0.85'; document.getElementById('mealsPerDay').value = pet.mealsPerDay ?? '3'; updateSpeciesButtonsUI(); populateStages(); if (pet.stageId) { const stageSelect = document.getElementById('petStage'); const matched = Array.from(stageSelect.options).find(opt => opt.dataset.id === pet.stageId); if (matched) { stageSelect.value = matched.value; } } lastAgeMonths = getTotalAgeMonths(); updateFoodRatioLabel(); updatePetNameDisplay(false); renderProfileChips(); calculatePortion(false); } // Add a new pet profile function addNewPetProfile() { const nextIndex = petStore.pets.length + 1; const newPet = createDefaultPet(`น้องตัวที่ ${nextIndex}`); petStore.pets.push(newPet); loadPetIntoForm(newPet.id); } // Delete the currently selected pet profile function deleteCurrentPetProfile() { if (petStore.pets.length <= 1) return; const activePet = getActivePet(); const displayName = (activePet.petName || '').trim() || 'สัตว์เลี้ยงตัวนี้'; const confirmed = confirm(`ต้องการลบโปรไฟล์ของ "${displayName}" ใช่หรือไม่?`); if (!confirmed) return; petStore.pets = petStore.pets.filter(p => p.id !== activePet.id); const nextPet = petStore.pets[0]; loadPetIntoForm(nextPet.id); } // Render the horizontal profile switcher chips function renderProfileChips() { const container = document.getElementById('petProfileList'); const deleteBtn = document.getElementById('btnDeleteProfile'); if (!container) return; // แสดงปุ่มลบเฉพาะเมื่อมีสัตว์เลี้ยงมากกว่า 1 ตัว if (deleteBtn) { if (petStore.pets.length > 1) { deleteBtn.classList.remove('hidden'); } else { deleteBtn.classList.add('hidden'); } } container.innerHTML = ''; petStore.pets.forEach(pet => { const isActive = pet.id === petStore.activeId; const name = (pet.petName || '').trim() || 'ไม่มีชื่อ'; const weight = parseFloat(pet.petWeight); const weightLabel = !isNaN(weight) && weight > 0 ? `${weight} กก.` : ''; const iconName = pet.species === 'dog' ? 'dog' : 'cat'; const btn = document.createElement('button'); btn.type = 'button'; btn.onclick = () => { if (pet.id !== petStore.activeId) { loadPetIntoForm(pet.id); } }; if (isActive) { btn.className = 'shrink-0 px-3.5 py-2 rounded-2xl text-xs font-semibold flex items-center gap-2 transition bg-gradient-to-r from-purple-600 to-purple-500 text-white shadow-sm shadow-purple-200'; } else { btn.className = 'shrink-0 px-3.5 py-2 rounded-2xl text-xs font-medium flex items-center gap-2 transition bg-purple-50/80 text-purple-800 border border-purple-100 hover:bg-purple-100/70'; } btn.innerHTML = ` ${escapeHTML(name)} ${weightLabel ? `${weightLabel}` : ''} `; container.appendChild(btn); }); lucide.createIcons(); } // Simple HTML escaper for safe rendering of pet names function escapeHTML(str) { return str.replace(/[&<>"']/g, match => ({ '&': '&', '<': '<', '>': '>', '"': '"', "'": ''' }[match])); } // Update Species Tab Button Styles function updateSpeciesButtonsUI() { const btnCat = document.getElementById('btnCat'); const btnDog = document.getElementById('btnDog'); if (currentSpecies === 'cat') { btnCat.className = 'py-2.5 px-4 rounded-xl font-medium text-sm flex items-center justify-center gap-2 transition tab-active'; btnDog.className = 'py-2.5 px-4 rounded-xl font-medium text-sm flex items-center justify-center gap-2 transition text-purple-700 hover:bg-purple-100/60'; } else { btnDog.className = 'py-2.5 px-4 rounded-xl font-medium text-sm flex items-center justify-center gap-2 transition tab-active'; btnCat.className = 'py-2.5 px-4 rounded-xl font-medium text-sm flex items-center justify-center gap-2 transition text-purple-700 hover:bg-purple-100/60'; } } // Calculate total age in months function getTotalAgeMonths() { const years = parseInt(document.getElementById('petAgeYears').value) || 0; const months = parseInt(document.getElementById('petAgeMonths').value) || 0; return (years * 12) + months; } // Auto-select stage based on species and age function autoSelectStageByAge(force = false) { const totalMonths = getTotalAgeMonths(); if (!force && totalMonths === lastAgeMonths) return; lastAgeMonths = totalMonths; let targetStageId = ''; if (currentSpecies === 'cat') { if (totalMonths < 12) { targetStageId = 'cat_kitten'; } else if (totalMonths >= 84) { targetStageId = 'cat_senior'; } else { targetStageId = 'cat_neutered'; } } else { if (totalMonths < 4) { targetStageId = 'dog_puppy_4m'; } else if (totalMonths < 12) { targetStageId = 'dog_puppy_4m_plus'; } else if (totalMonths >= 84) { targetStageId = 'dog_senior'; } else { targetStageId = 'dog_neutered'; } } const stageSelect = document.getElementById('petStage'); const options = Array.from(stageSelect.options); const matchedOption = options.find(opt => opt.dataset.id === targetStageId); if (matchedOption) { stageSelect.value = matchedOption.value; } } // Update Pet Name across UI function updatePetNameDisplay(shouldSave = true) { const name = document.getElementById('petName').value.trim() || 'สัตว์เลี้ยงของคุณ'; document.getElementById('lockPetName').textContent = name; document.getElementById('resPetName').textContent = name; if (shouldSave) { syncCurrentFormToActivePet(); } } // Switch between Cat & Dog function selectSpecies(species) { currentSpecies = species; updateSpeciesButtonsUI(); populateStages(); autoSelectStageByAge(true); calculatePortion(true); } // Populate Dropdown Options function populateStages() { const stageSelect = document.getElementById('petStage'); stageSelect.innerHTML = ''; const list = currentSpecies === 'cat' ? catStages : dogStages; list.forEach(item => { const opt = document.createElement('option'); opt.value = item.factor; opt.dataset.id = item.id; opt.textContent = item.label; stageSelect.appendChild(opt); }); } // Update Dry/Wet Percentage Label function updateFoodRatioLabel() { const dryVal = document.getElementById('mainFoodRatio').value; const wetVal = 100 - dryVal; document.getElementById('dryRatioVal').textContent = dryVal + '%'; document.getElementById('wetRatioVal').textContent = wetVal + '%'; } // Main Calorie & Portion Calculation function calculatePortion(shouldSave = true) { autoSelectStageByAge(false); if (shouldSave) { syncCurrentFormToActivePet(); } const isConfirmed = document.getElementById('healthConfirmed').checked; const notice = document.getElementById('unconfirmedNotice'); const resultCard = document.getElementById('resultCard'); if (!isConfirmed) { notice.classList.remove('hidden'); resultCard.classList.add('hidden'); return; } else { notice.classList.add('hidden'); resultCard.classList.remove('hidden'); } const weight = parseFloat(document.getElementById('petWeight').value) || 0; if (weight <= 0) return; const factor = parseFloat(document.getElementById('petStage').value) || 1.2; // RER = 70 * (weight ^ 0.75) const rer = 70 * Math.pow(weight, 0.75); const der = rer * factor; // 90% Main Food, 10% Treats const mainFoodCal = der * 0.90; const treatCal = der * 0.10; // Main Food Ratio Split (Dry vs Wet) const dryRatioPct = parseFloat(document.getElementById('mainFoodRatio').value) / 100; const wetRatioPct = 1 - dryRatioPct; const dryCalTarget = mainFoodCal * dryRatioPct; const wetCalTarget = mainFoodCal * wetRatioPct; // Densities const dryDensity = parseFloat(document.getElementById('dryCal').value) || 3.8; const wetDensity = parseFloat(document.getElementById('wetCal').value) || 0.85; const dryGrams = dryCalTarget / dryDensity; const wetGrams = wetCalTarget / wetDensity; // Meals const meals = parseInt(document.getElementById('mealsPerDay').value) || 3; const dryPerMeal = dryGrams / meals; const wetPerMeal = wetGrams / meals; // Water intake estimation const waterFactor = currentSpecies === 'cat' ? 50 : 60; const waterMl = weight * waterFactor; // Render UI document.getElementById('resRER').textContent = Math.round(rer); document.getElementById('resDER').textContent = Math.round(der); document.getElementById('resMainCal').textContent = Math.round(mainFoodCal); document.getElementById('resTreatCal').textContent = Math.round(treatCal); document.getElementById('resDryCal').textContent = Math.round(dryCalTarget); document.getElementById('resDryGram').textContent = dryGrams.toFixed(1); document.getElementById('resWetCal').textContent = Math.round(wetCalTarget); document.getElementById('resWetGram').textContent = wetGrams.toFixed(1); document.getElementById('resTreatMaxCal').textContent = Math.round(treatCal); document.getElementById('resMealCount').textContent = meals; document.getElementById('resDryPerMeal').textContent = dryPerMeal.toFixed(1); document.getElementById('resWetPerMeal').textContent = wetPerMeal.toFixed(1); document.getElementById('resWater').textContent = Math.round(waterMl); } // Initial setup on page load initPetStore(); loadPetIntoForm(petStore.activeId);
